{% if lang == 'zh' %}AIO 三层调度器根据需求预测自动伸缩工具实例，透明地吸收沙箱执行的延迟尖峰。{% else %}AIO three-layer scheduler auto-scales tool instances based on demand prediction. Absorbs latency spikes from sandbox execution transparently.{% endif %}

{% if lang == 'zh' %}可插拔 AgentFlow{% else %}Pluggable AgentFlow{% endif %}

{% if lang == 'zh' %}快速开始{% else %}Quick Start{% endif %}

{% if lang == 'zh' %}三步启动{% else %}Get running in three steps{% endif %}

1

{% if lang == 'zh' %}安装{% else %}Install{% endif %}

# Clone and install
git clone <repo-url>
cd siirl-agentic
pip install -e ".[gpu]"

# Verify
python -c "import siirl"
2

{% if lang == 'zh' %}配置{% else %}Configure{% endif %}

# config.yaml
algorithm: grpo
model:
  name: Qwen/Qwen2.5-7B
  tp_size: 4
rollout:
  max_turns: 8
3

{% if lang == 'zh' %}启动{% else %}Launch{% endif %}

# Start training
python -m siirl.async_train \
  --config config.yaml \
  --rollout-workers 4 \
  --tool-instances 16
